using System.Collections.Generic;
using UnityEngine;

public class MaskToEdgeColliders : MonoBehaviour
{
    [Header("Configuración")]
    public Sprite maskSprite;
    public string platformLayerName = "Platform";
    public bool isOneWay = true;

    [ContextMenu("Generar Colisiones Limpias")]
    public void GenerateColliders()
    {
        if (maskSprite == null) return;

        int targetLayer = LayerMask.NameToLayer(platformLayerName);
        if (targetLayer == -1) targetLayer = 0;

        // Limpiar colisionadores viejos
        EdgeCollider2D[] oldEdges = GetComponents<EdgeCollider2D>();
        for (int i = oldEdges.Length - 1; i >= 0; i--) DestroyImmediate(oldEdges[i]);
        
        PolygonCollider2D[] oldPolys = GetComponents<PolygonCollider2D>();
        for (int i = oldPolys.Length - 1; i >= 0; i--) DestroyImmediate(oldPolys[i]);

        int shapeCount = maskSprite.GetPhysicsShapeCount();

        for (int i = 0; i < shapeCount; i++)
        {
            List<Vector2> path = new List<Vector2>();
            maskSprite.GetPhysicsShape(i, path);

            if (path.Count < 2) continue;

            // Convertimos la silueta en un EdgeCollider2D (Línea caminable)
            EdgeCollider2D edge = gameObject.AddComponent<EdgeCollider2D>();
            edge.points = path.ToArray();

            if (isOneWay)
            {
                edge.usedByEffector = true;
            }
        }

        gameObject.layer = targetLayer;

        if (isOneWay)
        {
            PlatformEffector2D effector = GetComponent<PlatformEffector2D>();
            if (effector == null) effector = gameObject.AddComponent<PlatformEffector2D>();
            effector.useOneWay = true;
            effector.surfaceArc = 140f; // Ajustado para evitar atascos en bordes
        }

        Debug.Log($"¡Generados {shapeCount} EdgeColliders caminables perfectamente!");
    }
}
